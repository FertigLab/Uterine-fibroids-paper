SnC vs non-SnC Findmarkers result thresholded on p-value and FC

de_genes_list <- lapply(SnC_clusters, function(clust) {
  SnC.genes <- read.csv(here::here("results", "2_SnC-vs-nonSnC", tissue, paste0(tissue, ".", clust, ".SnC.genes.csv")), row.names = 1)
  SnC.genes <- SnC.genes %>%
    filter(p_val_adj < 0.05 & abs(avg_log2FC) > 0.3) %>%
    rownames_to_column()
})
names(de_genes_list) <- SnC_clusters
de_genes_interesting <- list_rbind(de_genes_list, names_to = "cell_type") 
de_genes_interesting$cell_type <- paste0("SnC.", de_genes_interesting$cell_type)
