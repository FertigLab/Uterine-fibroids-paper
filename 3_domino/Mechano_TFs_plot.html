tf_vln <- function(dom, regulon, plot_clusters = NULL, cols = NULL) {
  auc_mat <- get_tf_auc(dom, regulon) %>%
    filter(cluster %in% plot_clusters)
  auc_mat$cluster <- droplevels(auc_mat$cluster)
  auc_mat.long <- auc_mat %>%
    pivot_longer(cols = colnames(auc_mat)[-which(colnames(auc_mat) == "cluster")], names_to = "TF", values_to = "AUC")
  auc_mat.long$TF <- factor(auc_mat.long$TF, levels = regulon)
  plot <- ggplot(auc_mat.long, aes(x = cluster, y = AUC, col = cluster, fill = cluster)) + 
    geom_violin(trim = TRUE, scale = "width", adjust = 1) + 
    #geom_beeswarm(size = 0.1, method = "compactswarm", corral = "random", corral.width = 0.9) +
    #geom_point(aes(x = cluster, y = get(regulon)), position = "jitter", size = 0.5) + 
    facet_grid(rows = "TF") + 
    theme_classic() + theme(axis.text.x = element_text(angle = 90), legend.position = "none") + 
    labs(x = NULL, y = "AUC") 
  if(! is.null(cols)) {
    myplot <- plot + scale_color_manual(values = cols) + scale_fill_manual(values = cols)
    return(myplot)
  }
  else {
    return(plot)
  }
}
