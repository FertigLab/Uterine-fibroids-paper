Barplot function

barplot <- function(genes, title) {
  df <- as.data.frame(genes)
  colnames(df) <- "Avg_Exp"
  df$gene <- factor(rownames(df), levels = rownames(df))

  ggplot(df, aes(x = gene, y = Avg_Exp)) + geom_col(fill = "#A8554EFF") + theme_pubr() + theme(axis.text.x = element_text(angle = 45, vjust = 1, hjust = 1, size = 11)) + labs(x = NULL, title = title) 
}
